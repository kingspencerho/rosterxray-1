# Step 1: Identify "Naked" Status during the Stack-Building Loop (~Line 2116)
def evaluate_roster_stacks(roster):
    for player in roster:
        if player.position == "RB":
            # Check if this RB belongs to an existing team stack/loop
            is_stacked = check_if_team_has_qb_or_pass_catcher_stack(player.team, roster)
            
            if not is_stacked:
                player.type = "Naked"
                player.hasQB = False
                # Trigger the insulation evaluation flag
                player.requires_insulation_check = True
            else:
                player.type = "Stacked"
                player.requires_insulation_check = False


# Step 2: Enforce HVT / Zone-PROE Thresholds during Scoring (~Line 2700)
def score_naked_rb_insulation(player):
    if not player.requires_insulation_check:
        return # Skip stacked RBs

    # Define thresholds based on existing player/situation metrics
    # HVT = High-Value Touches (Targets + Green Zone Carries per game)
    # Zone_PROE = Composite score of team zone scheme fit and positive PROE
    
    HVT_THRESHOLD = 4.5  # Elite baseline for standalone volume
    SCHEME_THRESHOLD = 0.65  # Verified elite zone/PROE infrastructure environment

    has_elite_hvt = player.metrics.high_value_touches >= HVT_THRESHOLD
    has_elite_scheme = player.situation.scheme_proe_score >= SCHEME_THRESHOLD

    # The Insulation Logic Gate (Must hit 1 of 2 criteria)
    if not (has_elite_hvt or has_elite_scheme):
        # Push structural weakness note to the scoring engine output
        player.scoring_weaknesses.append({
            "code": "UNINSULATED_NAKED_RB",
            "message": f"WARNING: {player.name} is an uninsulated Naked RB. Fails both baseline High-Value Touch ({player.metrics.high_value_touches}) and Zone/PROE Scheme ({player.situation.scheme_proe_score}) thresholds. Roster ceiling severely capped."
        })
        
        # Apply a minor structural penalty to the total roster/construction score
        player.roster_construction_multiplier *= 0.90
